cmake_minimum_required(VERSION 3.25)
project(cpp_systems_tools VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
include(CTest)
find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
set(SANITIZER "" CACHE STRING "address, undefined, or thread")
add_compile_options(-Wall -Wextra -Wpedantic)
if(SANITIZER)
  add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
  add_link_options(-fsanitize=${SANITIZER})
endif()
add_subdirectory(projects/tcp-kv)
add_subdirectory(projects/runtime-inspector)
if(BUILD_TESTING)
  add_test(NAME evidence-scripts COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_scripts.py)
  set_tests_properties(evidence-scripts PROPERTIES TIMEOUT 15)
  add_test(NAME memory-evidence COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/test_memory_evidence.py)
  set_tests_properties(memory-evidence PROPERTIES LABELS memory TIMEOUT 15)
endif()
