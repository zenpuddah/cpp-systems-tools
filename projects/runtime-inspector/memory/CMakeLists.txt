cmake_minimum_required(VERSION 3.25)
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
  project(runtime_memory LANGUAGES CXX)
  set(CMAKE_CXX_STANDARD 20)
  set(CMAKE_CXX_STANDARD_REQUIRED ON)
  set(CMAKE_CXX_EXTENSIONS OFF)
  set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
  include(CTest)
  find_package(Python3 3.10 REQUIRED COMPONENTS Interpreter)
  add_compile_options(-Wall -Wextra -Wpedantic)
  set(SANITIZER "" CACHE STRING "address or undefined")
  if(SANITIZER)
    add_compile_options(-fsanitize=${SANITIZER} -fno-omit-frame-pointer -g)
    add_link_options(-fsanitize=${SANITIZER})
  endif()
endif()
add_library(runtime-memory-lib src/pmr.cpp src/experiments.cpp)
target_include_directories(runtime-memory-lib PUBLIC include)
target_compile_features(runtime-memory-lib PUBLIC cxx_std_20)
add_executable(runtime-memory src/main.cpp)
target_link_libraries(runtime-memory PRIVATE runtime-memory-lib)
if(BUILD_TESTING)
  add_executable(runtime-memory-core tests/core.cpp)
  target_include_directories(runtime-memory-core PRIVATE include)
  target_compile_features(runtime-memory-core PRIVATE cxx_std_20)
  foreach(area layout arena pool)
    add_test(NAME memory-${area} COMMAND runtime-memory-core ${area})
    set_tests_properties(memory-${area} PROPERTIES LABELS memory TIMEOUT 20)
  endforeach()
  add_executable(runtime-memory-pmr tests/pmr.cpp)
  target_link_libraries(runtime-memory-pmr PRIVATE runtime-memory-lib)
  add_test(NAME memory-pmr COMMAND runtime-memory-pmr)
  add_executable(runtime-memory-costs tests/costs.cpp)
  target_link_libraries(runtime-memory-costs PRIVATE runtime-memory-lib)
  add_test(NAME memory-costs COMMAND runtime-memory-costs)
  set_tests_properties(memory-pmr memory-costs PROPERTIES LABELS memory TIMEOUT 30)
  add_test(NAME memory-driver COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/tests/check_driver.py --binary $<TARGET_FILE:runtime-memory>)
  set_tests_properties(memory-driver PROPERTIES LABELS memory TIMEOUT 90)
endif()
